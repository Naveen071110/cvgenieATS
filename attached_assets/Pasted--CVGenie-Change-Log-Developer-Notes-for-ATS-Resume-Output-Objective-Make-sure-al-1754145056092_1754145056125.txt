def enforce_original_data(original, generated):
    # Simple check/replace for each vital field
    for field in ["name","address","phone","email"]:
        if original[field] not in generated:
            # Insert/replace in generated text as needed (can be improved with regex)
            generated = generated.replace(field.upper(), f"{field.upper()}:\n{original[field]}")
    # Optionally parse and check company, school, etc. in experience/education blocks

def format_to_ats(output):
    # Force all section headers and bullets to be clean, replace paragraphs
    output = re.sub(r'\n{2,}', '\n\n', output)
    headers = ["CONTACT INFORMATION", "SUMMARY", "WORK EXPERIENCE", "EDUCATION", "SKILLS"]
    for h in headers:
        output = output.replace(h, f"\n{h}")
