### 5. **Resume Data Functions**
Create `server/database/resumeQueries.ts` with these exact functions:
```typescript
import { pool } from './neon';

export interface Resume {
  id: number;
  user_id: string;
  resume_text: string;
  cover_letter?: string;
  job_description?: string;
  created_at: Date;
}

// Insert new resume into NEON POSTGRES ONLY
export async function insertResume(
  userId: string, 
  resumeText: string, 
  coverLetter?: string, 
  jobDescription?: string
): Promise<Resume> {
  const query = `
    INSERT INTO resumes (user_id, resume_text, cover_letter, job_description)
    VALUES ($1, $2, $3, $4)
    RETURNING *
  `;
  const values = [userId, resumeText, coverLetter, jobDescription];
  const result = await pool.query(query, values);
  return result.rows[0];
}

// Get all resumes for user from NEON POSTGRES ONLY
export async function getResumesByUserId(userId: string): Promise<Resume[]> {
  const query = `
    SELECT * FROM resumes 
    WHERE user_id = $1 
    ORDER BY created_at DESC
  `;
  const result = await pool.query(query, [userId]);
  return result.rows;
}
```

### 6. **Express API Endpoints**
Add to `server/routes.ts`:
```typescript
import { insertResume, getResumesByUserId } from './database/resumeQueries';
import { ClerkExpressRequireAuth } from '@clerk/clerk-sdk-node';

// POST /api/resume-history - Save resume to NEON POSTGRES ONLY
app.post('/api/resume-history', ClerkExpressRequireAuth(), async (req, res) => {
  try {
    const userId = req.auth.userId;
    const { resumeText, coverLetter, jobDescription } = req.body;
    
    // SAVE TO NEON POSTGRES EXTERNAL DATABASE ONLY
    const resume = await insertResume(userId, resumeText, coverLetter, jobDescription);
    
    res.json({ success: true, resume });
  } catch (error) {
    console.error('Error saving to Neon database:', error);
    res.status(500).json({ error: 'Failed to save resume to external database' });
  }
});

// GET /api/resume-history - Fetch from NEON POSTGRES ONLY
app.get('/api/resume-history', ClerkExpressRequireAuth(), async (req, res) => {
  try {
    const userId = req.auth.userId;
    
    // FETCH FROM NEON POSTGRES EXTERNAL DATABASE ONLY
    const resumes = await getResumesByUserId(userId);
    
    res.json({ resumes });
  } catch (error) {
    console.error('Error fetching from Neon database:', error);
    res.status(500).json({ error: 'Failed to fetch resumes from external database' });
  }
});
```

### 7. **Frontend Resume History Component**
Create `src/components/ResumeHistory.tsx`:
```typescript
import React, { useState, useEffect } from 'react';
import { useAuth } from '@clerk/clerk-react';

interface Resume {
  id: number;
  resume_text: string;
  cover_letter?: string;
  job_description?: string;
  created_at: string;
}

export const ResumeHistory: React.FC = () => {
  const { getToken } = useAuth();
  const [resumes, setResumes] = useState<Resume[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchResumes();
  }, []);

  const fetchResumes = async () => {
    try {
      const token = await getToken();
      const response = await fetch('/api/resume-history', {
        headers: { Authorization: `Bearer ${token}` }
      });
      const data = await response.json();
      setResumes(data.resumes);
    } catch (error) {
      console.error('Error fetching resume history:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="resume-history">
      <h2>My Resume History</h2>
      {loading ? (
        <p>Loading your resume history...</p>
      ) : resumes.length === 0 ? (
        <p>No resumes generated yet!</p>
      ) : (
        <div className="resume-list">
          {resumes.map((resume) => (
            <div key={resume.id} className="resume-card">
              <h3>Resume #{resume.id}</h3>
              <p>Created: {new Date(resume.created_at).toLocaleDateString()}</p>
              <button onClick={() => window.open(`data:text/plain,${encodeURIComponent(resume.resume_text)}`)}>
                Download Resume
              </button>
              {resume.cover_letter && (
                <button onClick={() => window.open(`data:text/plain,${encodeURIComponent(resume.cover_letter)}`)}>
                  Download Cover Letter
                </button>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
```
