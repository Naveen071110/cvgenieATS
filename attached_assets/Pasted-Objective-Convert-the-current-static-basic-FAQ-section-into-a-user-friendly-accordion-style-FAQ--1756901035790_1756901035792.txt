Accordion Functionality:
Add JavaScript to handle toggling open/close of FAQ items and maintaining ARIA attributes for accessibility, for example:

js
const faqItems = document.querySelectorAll('.faq-item button.faq-question');

faqItems.forEach(button => {
  button.addEventListener('click', () => {
    const faqItem = button.parentElement;
    const isOpen = faqItem.classList.contains('open');

    // Close all items
    document.querySelectorAll('.faq-item.open').forEach(item => {
      item.classList.remove('open');
      item.querySelector('.faq-question').setAttribute('aria-expanded', 'false');
    });

    // Toggle current item
    if (!isOpen) {
      faqItem.classList.add('open');
      button.setAttribute('aria-expanded', 'true');
    }
  });
});
Search Functionality:
Implement real-time filtering of FAQ questions as the user types in the search input:

js
const searchInput = document.querySelector('.faq-search-input');
const faqItemsContainer = document.querySelector('.faq-accordion');
const faqItemsArray = Array.from(faqItemsContainer.children);

searchInput.addEventListener('input', () => {
  const query = searchInput.value.toLowerCase();

  faqItemsArray.forEach(item => {
    const questionText = item.querySelector('.faq-question span').textContent.toLowerCase();
    if (questionText.includes(query)) {
      item.style.display = '';
    } else {
      item.style.display = 'none';
    }
  });
});
