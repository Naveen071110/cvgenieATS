// services/documentParser.js
const mammoth = require('mammoth');
const fs = require('fs');

class DocumentParser {
  async extractText(filePath, mimeType) {
    switch (mimeType) {
      case 'application/pdf':
        return await this.extractFromPDF(filePath);
        
      case 'application/vnd.openxmlformats-officedocument.wordprocessingml.document':
        return await this.extractFromDOCX(filePath);
        
      case 'text/plain':
        return await this.extractFromTXT(filePath);
        
      default:
        throw new Error('Unsupported file format');
    }
  }
  
  async extractFromDOCX(filePath) {
    try {
      const result = await mammoth.extractRawText({ path: filePath });
      return {
        content: result.value,
        wordCount: result.value.split(/\s+/).length,
        isValid: result.value.length > 100
      };
    } catch (error) {
      throw new Error(`Failed to extract DOCX content: ${error.message}`);
    }
  }
  
  async extractFromTXT(filePath) {
    try {
      const content = fs.readFileSync(filePath, 'utf-8');
      return {
        content: content,
        wordCount: content.split(/\s+/).length,
        isValid: content.length > 50
      };
    } catch (error) {
      throw new Error(`Failed to read TXT file: ${error.message}`);
    }
  }
  
  // Keep existing extractFromPDF method
}

module.exports = new DocumentParser();
4. Export Functionality Implementation
Document Generator Service:

javascript
// services/documentGenerator.js
const PDFDocument = require('pdfkit');
const officegen = require('officegen');
const fs = require('fs');

class DocumentGenerator {
  async generatePDF(content, outputPath) {
    const doc = new PDFDocument();
    doc.pipe(fs.createWriteStream(outputPath));
    
    // Add content formatting for resume
    doc.fontSize(16).text('Resume', 50, 50);
    doc.fontSize(12).text(content, 50, 80, {
      width: 500,
      align: 'left'
    });
    
    doc.end();
    return outputPath;
  }
  
  async generateDOCX(content, outputPath) {
    const docx = officegen('docx');
    
    // Add content to document
    const pObj = docx.createP();
    pObj.addText(content);
    
    return new Promise((resolve, reject) => {
      const out = fs.createWriteStream(outputPath);
      
      out.on('error', reject);
      out.on('close', () => resolve(outputPath));
      
      docx.generate(out);
    });
  }
  
  async generateMultipleFormats(resumeContent, coverLetterContent, baseFilename) {
    const outputs = {};
    
    // Generate PDF versions
    outputs.resumePDF = await this.generatePDF(resumeContent, `${baseFilename}_resume.pdf`);
    outputs.coverLetterPDF = await this.generatePDF(coverLetterContent, `${baseFilename}_cover_letter.pdf`);
    
    // Generate DOCX versions
    outputs.resumeDOCX = await this.generateDOCX(resumeContent, `${baseFilename}_resume.docx`);
    outputs.coverLetterDOCX = await this.generateDOCX(coverLetterContent, `${baseFilename}_cover_letter.docx`);
    
    return outputs;
  }
}

module.exports = new DocumentGenerator();
