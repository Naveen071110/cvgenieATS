Functional requirements for demo form:

Live text input with debounce to simulate asynchronous AI suggestion updates.

Display a section highlighting smart keyword additions or formatting improvements dynamically.

Show an animated ATS score bar that adjusts as mock input changes.

Example structure:

jsx
function DemoForm() {
  const [input, setInput] = React.useState('');
  const [atsScore, setAtsScore] = React.useState(45);
  const [improvements, setImprovements] = React.useState([]);

  React.useEffect(() => {
    // Simulate AI update with debounce
    const timeout = setTimeout(() => {
      setImprovements(generateMockImprovements(input));
      setAtsScore(calculateMockScore(input));
    }, 500);

    return () => clearTimeout(timeout);
  }, [input]);

  return (
    <div>
      <textarea value={input} onChange={e => setInput(e.target.value)} />
      <div className="ats-score-bar" style={{ width: `${atsScore}%` }}>
        ATS Score: {atsScore}%
      </div>
      <ul>
        {improvements.map((imp, idx) => (
          <li key={idx}>{imp}</li>
        ))}
      </ul>
    </div>
  );
}
